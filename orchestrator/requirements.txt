# Web framework
fastapi==0.115.0          # The API framework. Chosen over Flask because it's async-native
                          # (critical when you're waiting on 3 MCP servers simultaneously)
                          # and it auto-generates OpenAPI docs at /docs.
uvicorn[standard]==0.32.0 # ASGI server — the thing that actually runs FastAPI.
                          # Think of FastAPI as the app, uvicorn as the web server.
                          # [standard] includes websocket + hot-reload support.

# Data validation
pydantic[email]==2.8.2    # How FastAPI validates request/response shapes. You define a
                          # Python class, Pydantic ensures incoming JSON matches it.
                          # Pinned to 2.8.x — mcp==1.3.0 breaks with pydantic 2.9+
                          # (AnyUrl host_required constraint changed).
pydantic-settings==2.6.0  # Lets you load settings from .env files into a typed Pydantic
                          # class. Much cleaner than os.getenv() scattered everywhere.

# MCP client
mcp==1.3.0                # The official Anthropic MCP Python SDK. We use the client half
                          # to connect to calendar-mcp, whatsapp-mcp, and health-mcp.
httpx==0.28.0             # Async HTTP client. Used internally by the MCP SDK for SSE
                          # connections. Also useful for any direct HTTP calls we need.

# RAG pipeline 
chromadb[default]==0.6.0  # Vector database

# Authentication
python-jose[cryptography]==3.3.0  # JWT creation and verification
passlib[bcrypt]==1.7.4            # Password hashing

# Utilities
python-dotenv==1.0.1      # Loads .env file into environment variables on startup.
